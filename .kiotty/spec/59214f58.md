## 책임
구현 내부에서 주고받는 레코드 구조체. 문자열을 std::string 으로 직접 소유해 수명 문제를 만들지 않는다. C 구조체로의 변환은 cursor 가 한다.

## 모듈 인터페이스
```cpp
namespace kiotty::core {
// 구현 내부에서 쓰는 레코드. 문자열을 직접 소유하며, C 구조체로의 변환은 cursor.h 가 한다.
struct ItemRecord {
    std::string id;
    KiottySpecItemKind kind{};
    KiottySpecItemCategory category = KiottySpecItemCategory_None;
    std::string parentId;  // 비어 있으면 상위 없음
    int32_t seq = 0;
    std::string title;
    std::string body;
    int64_t cpSeq = 0;     // 이 버전이 속한 체크포인트 (kPending 이면 아직 미기록)
    int64_t wseq = 0;      // 전역 쓰기 순번. 구현 상태 계산에 쓴다
    bool removed = false;
};
struct LinkRecord {
    std::string fromId;
    std::string toId;
    KiottySpecLinkType type{};
    int64_t cpSeq = 0;
};
struct FileRecord {
    std::string id;
    KiottySpecFileRole role{};
    std::string path;
    std::string hash;
    int64_t cpSeq = 0;
};
struct CheckpointRecord {
    int64_t seq = 0;
    int32_t step = 0;
    bool confirmed = false;
    KiottySpecCheckpointPurpose purpose = KiottySpecCheckpointPurpose_None;
    std::string note;
    int64_t createdAt = 0;
};
struct ChangeRequestRecord {
    std::string id;
    KiottySpecChangeSource source{};
    std::string body;
    int32_t entryStep = 0;
    KiottySpecChangeState state{};
    int64_t cpOpened = 0;
    std::optional<int64_t> cpClosed;
    std::string note;
};
struct ResultRecord {
    std::string targetId;
    KiottySpecTestResult result{};
    std::string log;
    int64_t cpSeq = 0;
    int64_t runAt = 0;
};
struct MilestoneEntryRecord {
    std::string milestoneId;
    std::string itemId;
    KiottySpecImplState state = KiottySpecImplState_NotImplemented;
    std::optional<int64_t> implementedCpSeq;
};
struct StepRunRecord {
    int32_t step = 0;
    int64_t firstSeq = 0;
    int64_t lastSeq = 0;
    std::optional<int64_t> confirmedSeq;
    size_t count = 0;
    std::string lastNote;
    std::string reason;
};
struct PositionRecord {
    std::optional<int64_t> checkpointSeq;
    int32_t step = 0;
    bool confirmed = false;
    bool released = false;
    size_t pendingWrites = 0;
    int32_t nextStep = 1;
    std::string nextAction;
};
} // namespace kiotty::core
```
